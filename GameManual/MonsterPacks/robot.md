怪物卡{
    名字: 人工智能核心领袖
    怪物级别: 首领
    怪物类型: "机器人"
    最大生命值: 19
    初始生命值: 19
    攻击伤害: 4
    射程: "中距离" # 攻击 1 和 2 位置上的所有玩家
    技能: {
        技能名: "人工智能核心领袖"
        技能描述: "当此怪物卡被玩家抓取时，抓取该卡的玩家弃掉所有手牌区的牌，并对射程内的所有玩家造成4点伤害。"
        skillType: "Monster"
        trigger: 怪物卡进入求生者怪物区后
        forced: true # 强制发动
        filter: true
        content: {
            抓取者 = event.玩家
            手牌 = getCard(抓取者, position = "手牌区") # 获取手牌区所有牌
            抓取者.discard(手牌) # 弃置所有手牌（进入游戏牌弃牌堆）
            List = getTargetInRange(self.纠缠对象, "中距离") # 以纠缠玩家所在位置为中心、中距离内的所有玩家
            for p in List:
                p.damage(4, self) # 造成4点来自此怪物的伤害
        }
    }
}

    初始生命值: 10

    技能: {
        技能名: "激光无人机-监视"
        技能描述: "被此怪物卡纠缠的玩家无法进行潜行检定，直接抓取怪物卡。"
        skillType: "Monster"
        trigger: 潜行检定前 # 玩家进入有怪物标记的地块、即将进行潜行检定时触发（见 E_gameJudge.md）

            event.玩家.drawMonster(1) # 抓取一张怪物卡
        }
    }

怪物卡{
    名字: 爆破机器人
    怪物级别: 精英
    怪物类型: "机器人"
    最大生命值: 15
    初始生命值: 15
    攻击伤害: 4
    射程: "无"
    技能1: {
        技能名: "爆破机器人-殉爆"
        技能描述: "当此怪物卡被消灭时，对它所纠缠的玩家造成2点伤害。"
        skillType: "Monster"
        trigger: 怪物死亡时
        forced: true
        filter: return event.target == self # 仅当此怪物自身被消灭时触发
        content: {
            self.纠缠对象.damage(2, self) # 对它所纠缠的玩家造成2点伤害
        }
    }
    技能2: {
        技能名: "爆破机器人-定点爆破"
        技能描述: "当此怪物卡造成伤害时，若受到伤害的玩家本回合没有移动过，造成的伤害翻倍。"
        skillType: "Monster"
        trigger: 造成伤害时 # 钩在伤害结算流程，修改伤害值变量
        forced: true
        filter: return event.source == self && event.target.countMark("本回合已移动") == 0 # 此怪物造成伤害，且受到伤害的玩家本回合没有移动过
        content: {
            event.num *= 2 # 造成的伤害翻倍
        }
    }
}
